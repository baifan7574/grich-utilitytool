def build():
    if not os.path.exists(SUBPAGE_DIR): os.makedirs(SUBPAGE_DIR)
    
    registry = []
    sitemap_urls = []
    count = 0
    
    with open(CSV_FILE, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            if count >= LIMIT_PAGES: # 【核心阀门】
                break
                
            p, s, st = row['profession'], row['slug'], row['state']
            
            # 智能匹配配色逻辑
            theme = THEME_CONFIG['Default']
            for key in THEME_CONFIG:
                if key.lower() in p.lower():
                    theme = THEME_CONFIG[key]
                    break
            
            # 生成子页 HTML
            pg = SUBPAGE_TEMPLATE.replace("{{title}}", f"{st} {p} Audit Tool")\
                              .replace("{{brand}}", BRAND_NAME)\
                              .replace("{{profession}}", p)\
                              .replace("{{state}}", st)\
                              .replace("{{theme_bg}}", theme['bg'])\
                              .replace("{{theme_text}}", theme['text'])\
                              .replace("{{theme_color}}", theme['color'])\
                              .replace("{{pay_link}}", PAYHIP_LINK)
            
            with open(os.path.join(SUBPAGE_DIR, f"{s}.html"), 'w', encoding='utf-8') as pf:
                pf.write(pg)
            
            registry.append({ "p": p, "slug": s, "st": st, "t_bg": theme['bg'], "t_text": theme['text'], "t_color": theme['color'] })
            sitemap_urls.append(f"{BASE_URL}/p/{s}.html")
            count += 1

    # 生成主页
    cards = ""
    for i in registry:
        cards += f'''<a href="p/{i['slug']}.html" class="card bg-white p-12 rounded-[3.5rem] border border-slate-100 shadow-sm hover:shadow-2xl transition-all text-left" data-s="{i['p']} {i['st']}"><div class="flex items-center gap-6 mb-8 text-left"><div class="w-16 h-16 {i['t_bg']} rounded-2xl flex items-center justify-center text-white font-black text-2xl shadow-lg">{i['p'][0]}</div><div><h3 class="font-black text-slate-900 text-xl leading-tight">{i['p']}</h3><p class="text-[11px] font-black text-slate-400 uppercase tracking-widest mt-1">{i['st']} Hub</p></div></div><div class="flex items-center text-sm font-black {i['t_text']} uppercase tracking-widest">Connect to Node -></div></a>'''
    
    final_idx = INDEX_TEMPLATE.split('{% for item in registry %}')[0].replace("{{brand}}", BRAND_NAME) + cards + INDEX_TEMPLATE.split('{% endfor %}')[1]
    with open(os.path.join(OUTPUT_DIR, "index.html"), 'w', encoding='utf-8') as f: f.write(final_idx)

    # 生成 Sitemap.xml
    today = datetime.now().strftime("%Y-%m-%d")
    sitemap = f'<?xml version="1.0" encoding="UTF-8"?>\\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\\n'
    sitemap += f'  <url><loc>{BASE_URL}/index.html</loc><lastmod>{today}</lastmod><priority>1.0</priority></url>\\n'
    for url in sitemap_urls:
        sitemap += f'  <url><loc>{url}</loc><lastmod>{today}</lastmod><priority>0.8</priority></url>\\n'
    sitemap += '</urlset>'
    with open(os.path.join(OUTPUT_DIR, "sitemap.xml"), 'w', encoding='utf-8') as f: f.write(sitemap)
    
    print(f"Success! Generated {count} pages and Sitemap.xml.")

if __name__ == "__main__":
    build()
